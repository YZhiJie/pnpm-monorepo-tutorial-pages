GitHub Actions 基础 ​

完整 CI 配置模板 ​

yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

env:
  NODE_VERSION: "18"
  PNPM_VERSION: "8"

jobs:
  setup:
    runs-on: ubuntu-latest
    outputs:
      cache-key: ${{ steps.cache-key.outputs.key }}
    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}

      - name: Setup pnpm
        uses: pnpm/action-setup@v2
        with:
          version: ${{ env.PNPM_VERSION }}

      - name: Get pnpm store directory
        id: pnpm-cache
        run: echo "STORE_PATH=$(pnpm store path)" >> $GITHUB_OUTPUT

      - name: Setup pnpm cache
        uses: actions/cache@v3
        with:
          path: ${{ steps.pnpm-cache.outputs.STORE_PATH }}
          key: ${{ runner.os }}-pnpm-store-${{ hashFiles('**/pnpm-lock.yaml') }}
          restore-keys: |
            ${{ runner.os }}-pnpm-store-

      - name: Generate cache key
        id: cache-key
        run: echo "key=${{ runner.os }}-${{ hashFiles('**/pnpm-lock.yaml') }}" >> $GITHUB_OUTPUT

  lint:
    needs: setup
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Run ESLint
        run: pnpm lint

      - name: Run Prettier
        run: pnpm format:check

  typecheck:
    needs: setup
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: TypeScript check
        run: pnpm typecheck

  test:
    needs: setup
    runs-on: ubuntu-latest
    strategy:
      matrix:
        node-version: [16, 18, 20]
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm
        with:
          node-version: ${{ matrix.node-version }}

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Run tests
        run: pnpm test

      - name: Upload coverage
        uses: codecov/codecov-action@v3
        if: matrix.node-version == 18

  build:
    needs: [lint, typecheck, test]
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Build all packages
        run: pnpm build

      - name: Upload artifacts
        uses: actions/upload-artifact@v3
        with:
          name: dist
          path: |
            apps/*/dist
            packages/*/dist

可复用 Action ​

yaml
# .github/actions/setup-pnpm/action.yml
name: "Setup pnpm"
description: "Setup Node.js, pnpm and cache"
inputs:
  node-version:
    description: "Node.js version"
    default: "18"

runs:
  using: composite
  steps:
    - name: Setup Node.js
      uses: actions/setup-node@v4
      with:
        node-version: ${{ inputs.node-version }}

    - name: Setup pnpm
      uses: pnpm/action-setup@v2
      with:
        version: 8

    - name: Get pnpm store directory
      id: pnpm-cache
      shell: bash
      run: echo "STORE_PATH=$(pnpm store path)" >> $GITHUB_OUTPUT

    - name: Setup pnpm cache
      uses: actions/cache@v3
      with:
        path: ${{ steps.pnpm-cache.outputs.STORE_PATH }}
        key: ${{ runner.os }}-pnpm-store-${{ hashFiles('**/pnpm-lock.yaml') }}
        restore-keys: |
          ${{ runner.os }}-pnpm-store-

构建流程自动化 ​

变更检测与增量构建 ​

yaml
- name: Detect changed packages
  id: changes
  uses: dorny/paths-filter@v2
  with:
    filters: |
      packages:
        - 'packages/**'
      apps:
        - 'apps/**'

- name: Build packages
  if: steps.changes.outputs.packages == 'true'
  run: pnpm --filter ./packages/* build

- name: Build apps
  if: steps.changes.outputs.apps == 'true'
  run: pnpm --filter ./apps/* build

并行构建 ​

yaml
strategy:
  matrix:
    shard: [1, 2, 3]
steps:
  - name: Build shard
    run: pnpm --filter "...[HEAD~1]" --shard ${{ matrix.shard }}/3 build

部署配置 ​

yaml
# .github/workflows/deploy.yml
name: Deploy

on:
  push:
    branches: [main, develop]
    tags: ["v*"]

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment:
      name: ${{ github.ref == 'refs/heads/main' && 'production' || 'staging' }}
      url: ${{ steps.deploy.outputs.url }}

    steps:
      - uses: actions/checkout@v4

      - name: Setup environment
        run: |
          if [[ "${{ github.ref }}" == "refs/heads/main" ]]; then
            echo "API_URL=https://api.example.com" >> $GITHUB_ENV
          else
            echo "API_URL=https://api-staging.example.com" >> $GITHUB_ENV
          fi

      - name: Deploy to Vercel
        id: deploy
        uses: amondnet/vercel-action@v25
        with:
          vercel-token: ${{ secrets.VERCEL_TOKEN }}
          vercel-org-id: ${{ secrets.VERCEL_ORG_ID }}
          vercel-project-id: ${{ secrets.VERCEL_PROJECT_ID }}
          vercel-args: ${{ github.ref == 'refs/heads/main' && '--prod' || '' }}

GitHub Pages 文档站部署 ​

教程站点通过 .github/workflows/deploy-pages.yml 自动部署到 GitHub Pages。每次 main 分支 push 都会触发构建与发布。

跨仓库方案（源码私有 + 站点公开） ​

GitHub 免费版不允许私有仓库开启 Pages，因此采用双仓库结构：

仓库可见性角色
pnpm-monorepo-tutorial私有源码（文档、练习、CI）
pnpm-monorepo-tutorial-pages公开部署产物（仅 gh-pages 分支）

main push 触发构建后，由 peaceiris/actions-gh-pages 把 .vitepress/dist 推送到部署仓的 gh-pages 分支（force_orphan，首推后 Pages 自动就绪），最终访问地址为 https://yzhijie.github.io/pnpm-monorepo-tutorial-pages/。

工作流要点 ​

  • 构建命令：pnpm docs:build，产物在 .vitepress/dist
  • 路径前缀：DEPLOY_BASE=/pnpm-monorepo-tutorial-pages/（project page 必须带仓库名前缀）
  • 认证：PAGES_TOKEN（对部署仓有 contents:write 权限的 PAT），与 frontend-advanced-guide 同一套模式
  • 降级：PAGES_TOKEN 未配置时跳过部署、保留构建验证，流水线不恒红

手动开启步骤（仓库管理员） ​

  1. 创建公开仓库 YZhiJie/pnpm-monorepo-tutorial-pages（空仓库即可）
  2. 在本仓库 Settings → Secrets and variables → Actions 配置 PAGES_TOKEN
  3. 下次 main push 即自动部署；部署仓的 Pages（gh-pages 分支）首推后自动就绪

若部署仓改名，同步修改 deploy-pages.yml 中的 DEPLOY_BASE 与 external_repository 两处。

部署前验证机制 ​

质量门禁 ​

yaml
# .github/workflows/quality-gate.yml
name: Quality Gate

on:
  pull_request:
    branches: [main]

jobs:
  quality:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Lint
        run: pnpm lint

      - name: Type check
        run: pnpm typecheck

      - name: Test
        run: pnpm test -- --coverage

      - name: Coverage threshold
        run: |
          COVERAGE=$(cat coverage/coverage-summary.json | jq '.total.lines.pct')
          if (( $(echo "$COVERAGE < 80" | bc -l) )); then
            echo "Coverage $COVERAGE% is below 80%"
            exit 1
          fi

      - name: Bundle size check
        run: |
          pnpm build
          SIZE=$(du -sb apps/web/dist | cut -f1)
          if [ $SIZE -gt 10485760 ]; then
            echo "Bundle size exceeds 10MB"
            exit 1
          fi

yaml
# .github/workflows/release.yml
name: Release

on:
  push:
    branches: [main]

jobs:
  release:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm

      - name: Create Release PR
        uses: changesets/action@v1
        with:
          version: pnpm version
          publish: pnpm release
          createGithubReleases: true
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          NPM_TOKEN: ${{ secrets.NPM_TOKEN }}
