GitHub Actions 基础 ​

完整 CI 配置模板 ​

yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

env:
  NODE_VERSION: "20"
  # 必须与根 package.json 的 packageManager 字段一致
  PNPM_VERSION: "9.15.4"

jobs:
  setup:
    runs-on: ubuntu-latest
    outputs:
      cache-key: ${{ steps.cache-key.outputs.key }}
    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}

      - name: Setup pnpm
        uses: pnpm/action-setup@v4
        with:
          version: ${{ env.PNPM_VERSION }}

      - name: Get pnpm store directory
        id: pnpm-cache
        run: echo "STORE_PATH=$(pnpm store path)" >> $GITHUB_OUTPUT

      - name: Setup pnpm cache
        uses: actions/cache@v4
        with:
          path: ${{ steps.pnpm-cache.outputs.STORE_PATH }}
          key: ${{ runner.os }}-pnpm-store-${{ hashFiles('**/pnpm-lock.yaml') }}
          restore-keys: |
            ${{ runner.os }}-pnpm-store-

      - name: Generate cache key
        id: cache-key
        run: echo "key=${{ runner.os }}-${{ hashFiles('**/pnpm-lock.yaml') }}" >> $GITHUB_OUTPUT

  lint:
    needs: setup
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Check dependency graph
        run: pnpm deps:check

      - name: Check version consistency
        run: pnpm version:check

      - name: Run ESLint
        run: pnpm lint

      - name: Run Prettier
        run: pnpm format:check

  typecheck:
    needs: setup
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: TypeScript check
        run: pnpm typecheck

  test:
    needs: setup
    runs-on: ubuntu-latest
    strategy:
      matrix:
        # Node 18 已 EOL；仓库实际测试矩阵为 20 / 22
        node-version: [20, 22]
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm
        with:
          node-version: ${{ matrix.node-version }}

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Run unit tests
        run: pnpm test:unit

      - name: Run integration tests
        run: pnpm test:integration

  coverage:
    # 阈值只对 utils / api 生效；web / ui 仍只产出报告
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Run coverage with per-package thresholds
        run: pnpm test:coverage

  build:
    needs: [lint, typecheck, test, coverage]
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./.github/actions/setup-pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Build all packages
        run: pnpm build

      - name: Upload artifacts
        uses: actions/upload-artifact@v4
        with:
          name: dist
          path: |
            apps/*/dist
            packages/*/dist

可复用 Action ​

yaml
# .github/actions/setup-pnpm/action.yml
name: "Setup pnpm"
description: "Setup Node.js, pnpm and cache"
inputs:
  node-version:
    description: "Node.js version"
    default: "20"
  turbo-cache:
    description: "是否恢复/保存 .turbo 缓存"
    default: "true"

runs:
  using: composite
  steps:
    - name: Setup Node.js
      uses: actions/setup-node@v4
      with:
        node-version: ${{ inputs.node-version }}

    - name: Setup pnpm
      uses: pnpm/action-setup@v4
      with:
        version: 9.15.4

    - name: Get pnpm store directory
      id: pnpm-cache
      shell: bash
      run: echo "STORE_PATH=$(pnpm store path)" >> $GITHUB_OUTPUT

    # pnpm store 与 Turbo 缓存分开缓存，key 维度不同、失效互不影响
    - name: Setup pnpm cache
      uses: actions/cache@v4
      with:
        path: ${{ steps.pnpm-cache.outputs.STORE_PATH }}
        key: ${{ runner.os }}-pnpm-store-${{ hashFiles('pnpm-lock.yaml') }}
        restore-keys: |
          ${{ runner.os }}-pnpm-store-

    - name: Resolve Turbo version
      if: inputs.turbo-cache == 'true'
      id: turbo-version
      shell: bash
      run: |
        version=$(grep -oE 'turbo@[0-9]+\.[0-9]+\.[0-9]+' pnpm-lock.yaml | head -1 | cut -d@ -f2)
        echo "version=${version:-unknown}" >> "$GITHUB_OUTPUT"

    - name: Setup Turbo cache
      if: inputs.turbo-cache == 'true'
      uses: actions/cache@v4
      with:
        path: .turbo
        key: ${{ runner.os }}-turbo-${{ hashFiles('pnpm-lock.yaml') }}-${{ steps.turbo-version.outputs.version }}
        restore-keys: |
          ${{ runner.os }}-turbo-${{ hashFiles('pnpm-lock.yaml') }}-

所有 CI 作业都通过 uses: ./.github/actions/setup-pnpm 复用这段逻辑，避免每个作业复制粘贴 setup/cache；不跑 Turbo 的作业传 turbo-cache: false。

自动化测试配置 ​

测试分层 ​

层承担包验证内容脚本
unitutils、ui、api纯函数、组件隔离渲染、环境配置 schematest:unit
integrationapi、ui、web真实 HTTP、dist 产物消费、整棵跨包组件树渲染test:integration
e2eweb真实浏览器下的首页、表单校验与跨包组件交互test:e2e（独立 CI job）

文件命名约定：单元测试保持 *.test.ts(x)；集成测试使用 *.integration.test.ts(x)，这样脚本可以用路径过滤，不需要拆分目录。

json
// packages/ui/package.json（web / api 只有 test:integration）
{
  "scripts": {
    "test": "pnpm test:unit && pnpm test:integration",
    "test:unit": "vitest run --exclude \"**/*.integration.test.*\"",
    "test:integration": "vitest run \"integration.test\"",
    "test:coverage": "vitest run --coverage --coverage.reporter=text --coverage.reporter=json-summary"
  }
}

api 的 test 同样先 unit 后 integration；config.test.ts 覆盖环境变量 schema，app.integration.test.ts 覆盖 /healthz、/readyz、/metrics 与 503 语义。

Turbo 任务：

json
{
  "test:unit": { "outputs": [] },
  "test:integration": { "dependsOn": ["build"], "outputs": [] },
  "test:coverage": { "dependsOn": ["build"], "outputs": ["coverage/**"] }
}
  • test 仍是汇总入口，dependsOn: ["build"] 的既有约定不变。
  • test:unit 只读源码，不触发构建；test:integration 要读 dist 或渲染整棵树，必须等 build。
  • CI 的 test 作业拆成 test:unit 与 test:integration 两个 step，并沿用 ...[origin/main] 过滤：
yaml
- name: Run unit tests
  run: |
    if [ -n "$TURBO_FILTER" ]; then
      pnpm exec turbo run test:unit --filter="$TURBO_FILTER"
    else
      pnpm test:unit
    fi

- name: Run integration tests
  run: |
    if [ -n "$TURBO_FILTER" ]; then
      pnpm exec turbo run test:integration --filter="$TURBO_FILTER"
    else
      pnpm test:integration
    fi

确定性约定：

  • api 集成测试用 app.listen(0) 随机端口 + 原生 fetch，beforeAll 启动、afterAll 关闭，不共享可变状态。
  • 测试之间无顺序依赖；不使用固定端口，避免 CI 并行时端口冲突。

覆盖率：按包逐步开启阈值 ​

bash
pnpm test:coverage
  • 使用 @vitest/coverage-v8，每个包输出 coverage/，其中 coverage-summary.json 可用于汇总。
  • 统计前先剔除非代码文件：apps/api 的进程入口 src/index.ts、apps/web 的 src/main.tsx、packages/ui 的 barrel src/index.ts，以及 Tailwind / PostCSS 配置，避免“从未被单元测试加载的入口”拉低分母。
  • 排除上述文件后的实测基线：
包行语句函数分支阈值
@tutorial/utils97.05%96.34%96%93.02%lines 95 / statements 94 / functions 94 / branches 91
@tutorial/api95.65%95.29%91.66%81.08%lines 93 / statements 93 / functions 88 / branches 79
@tutorial/ui20%19.35%11.11%16.66%不设（只出报告）
@tutorial/web25%22.72%12.5%33.33%不设（只出报告）
  • utils 只有三个纯逻辑文件，基线稳定；api 排除入口后，config.ts、app.ts 等核心逻辑都有测试，因此这两个包先开启阈值。
  • 阈值贴近基线并留约 2 个百分点余量；低于阈值时 vitest run --coverage 以非零码退出，CI 的 coverage 作业直接失败。
  • ui 的 Card / Input / ThemeProvider 还没有组件测试，当前数字主要由 Button 贡献；web 的交互主要交给 Playwright e2e，单测只覆盖组合与渲染。两者暂不设阈值，避免为了过门槛写低价值测试。
  • 负向实测：临时把 storage.test.ts 的三个 describe 改成 describe.skip，覆盖率降到行 66.17% / 语句 64.63% / 函数 64% / 分支 46.51%，Vitest 输出四条 ERROR: Coverage for ... does not meet global threshold 并以退出码 1 结束；恢复测试后重新通过。

ratchet 策略：

  1. 每次调整阈值时，先跑 pnpm test:coverage 记录当前基线，只允许在基线之上向上收紧；
  2. 单次收紧幅度不超过“基线 - 1 个百分点”，避免因一次重构或平台差异造成假失败；
  3. 覆盖率下降时先补测试或说明原因，不直接下调阈值；确实需要下调时，必须在提交信息里写明理由；
  4. 给 ui 设阈值的前提：Card / Input / ThemeProvider 至少各有一条行为测试，覆盖率连续 2 个迭代稳定，先设行覆盖阈值，再逐步加语句/函数/分支；
  5. web 同理，且要等单测覆盖范围从“只渲染首页”扩展到更多交互后再考虑。

浏览器层 e2e（Playwright） ​

e2e 只覆盖 apps/web 的三条关键路径，不重复测试 API 内部实现：

用例覆盖的交互
首页渲染标题、Card 卡片标题、Input 占位符、Button 初始禁用态
表单校验非法邮箱触发 role=alert 与 aria-invalid；合法邮箱解锁按钮并新增用户
跨包组件交互自动/手动从 /api/users 加载列表、日期格式化、清空列表与空态
bash
# 首次需要下载 Chromium（CI 上带系统依赖）
pnpm --filter @tutorial/web e2e:install

# 构建共享包与 api，再启动 web/api 并跑浏览器用例
pnpm test:e2e

设计取舍：

  • 不并入 pnpm test：e2e 依赖浏览器与两个常驻进程，耗时和失败原因都与 unit/integration 不同；默认测试保持快反馈，e2e 通过根脚本单独触发。
  • 只装 Chromium：三条用例都不依赖引擎差异，Firefox/WebKit 只会让下载与安装成本翻倍；需要跨引擎验证时再按需增加 project，而不是默认全装。
  • 固定端口 + strictPort：webServer 用 3000（api）与 5173（web），Vite 端口被占用时直接失败，避免自动换端口后探活地址失真。
  • 进程可回收：api 直接运行构建产物 apps/api/dist/index.js，不用 tsx watch；本地允许复用已有 dev server，CI 一律由 webServer 启动并在结束时清理。
  • 稳定性参数：retries: 1、trace: "on-first-retry"、失败时上传 playwright-report/ 与 test-results/，保留 7 天。
  • 确定性：固定 locale: "zh-CN" 与 timezoneId: "Asia/Shanghai"，让 formatDate 的断言不受 runner 时区影响。

CI 中是独立 job，固定在 Node 20 单点运行，不进 20/22 矩阵：

yaml
e2e:
  runs-on: ubuntu-latest
  steps:
    - uses: actions/checkout@v4
    - uses: ./.github/actions/setup-pnpm
      with:
        turbo-cache: false

    - name: Install dependencies
      run: pnpm install --frozen-lockfile

    # key 只随 Playwright 版本变化；版本不变时浏览器缓存命中
    - name: Resolve Playwright version
      id: playwright
      run: |
        version=$(node -p "require('./apps/web/node_modules/@playwright/test/package.json').version")
        echo "version=$version" >> "$GITHUB_OUTPUT"

    - name: Cache Playwright browsers
      uses: actions/cache@v4
      with:
        path: ~/.cache/ms-playwright
        key: ${{ runner.os }}-playwright-${{ steps.playwright.outputs.version }}
        restore-keys: |
          ${{ runner.os }}-playwright-

    - name: Install Chromium
      run: pnpm --filter @tutorial/web e2e:install

    - name: Run e2e tests
      run: pnpm test:e2e

    - name: Upload Playwright report
      if: failure()
      uses: actions/upload-artifact@v4
      with:
        name: playwright-report
        path: |
          apps/web/playwright-report
          apps/web/test-results
        retention-days: 7

成本与耗时（本地实测，仅作量级参考）：

  • 浏览器缓存命中时 playwright install chromium 约 1 秒；冷启动需要下载 Chromium（含 headless shell，约百 MB 量级），再叠加 --with-deps 的 apt 安装。
  • e2e 用例本身约 3 秒；加上共享包构建、api 构建与两个 dev server 启动，pnpm test:e2e 单次约 10 秒。
  • CI 上浏览器缓存命中时，该 job 预计比 unit/integration 多 1 分钟左右；冷启动还要下载 Chromium 并安装系统依赖，额外再多 2 分钟左右。

e2e 曾经“全绿但没真正跑起来”是常见坑，因此本批做过一次负向验证：

ts
// 临时用例，验完即删
test("负向验证：断言不存在的元素必须失败", async ({ page }) => {
  await page.goto("/");
  await expect(page.getByTestId("definitely-missing")).toBeVisible();
});

该用例按预期失败并触发重试、失败截图与 trace，说明报告、重试与产物链路都真实生效；随后删除，连续两次全量运行均以 3 passed 结束。

构建流程自动化 ​

变更检测与增量构建 ​

PR 只验证受影响包及其依赖方，push 到主分支则全量验证：

yaml
env:
  # PR 时只跑受影响包；push 时为空，表示全量
  TURBO_FILTER: ${{ github.event_name == 'pull_request' && '...[origin/main]' || '' }}

steps:
  - uses: actions/checkout@v4
    with:
      # 必须取全量历史，否则无法解析 origin/main 作为 diff 基线
      fetch-depth: 0
  - uses: ./.github/actions/setup-pnpm
  - run: pnpm install --frozen-lockfile

  - name: Build
    run: |
      if [ -n "$TURBO_FILTER" ]; then
        pnpm exec turbo run build --filter="$TURBO_FILTER"
      else
        pnpm build
      fi

--filter="...[origin/main]" 的语义是：自 origin/main 以来发生变更的包，以及依赖这些包的包（即依赖方）。等价 pnpm 写法：

bash
pnpm --filter "...[origin/main]" run build
pnpm exec turbo run typecheck --filter="...[origin/main]"

并行构建 ​

yaml
strategy:
  matrix:
    shard: [1, 2, 3]
steps:
  - name: Build shard
    run: pnpm --filter "...[HEAD~1]" --shard ${{ matrix.shard }}/3 build

容器镜像构建与推送 ​

ci.yml 的 docker 作业在 PR 上只做多架构构建验证，在 main 上且配置了 Docker Hub 凭据时才推送：

yaml
docker:
  needs: [build]
  runs-on: ubuntu-latest
  steps:
    - uses: actions/checkout@v4

    - uses: docker/setup-qemu-action@v3
    - uses: docker/setup-buildx-action@v3

    # secrets 不能直接写进 if，先通过 env 读入再判断
    - name: Resolve registry target
      id: registry
      env:
        DOCKERHUB_USERNAME: ${{ secrets.DOCKERHUB_USERNAME }}
        DOCKERHUB_TOKEN: ${{ secrets.DOCKERHUB_TOKEN }}
      run: |
        if [ "${{ github.event_name }}" = "push" ] \
          && [ "${{ github.ref }}" = "refs/heads/main" ] \
          && [ -n "$DOCKERHUB_USERNAME" ] \
          && [ -n "$DOCKERHUB_TOKEN" ]; then
          echo "push=true" >> "$GITHUB_OUTPUT"
          echo "image=${DOCKERHUB_USERNAME}/pnpm-monorepo-api" >> "$GITHUB_OUTPUT"
        else
          echo "push=false" >> "$GITHUB_OUTPUT"
          echo "::notice::PR 构建或未配置 Docker Hub 凭据，只构建不推送"
        fi

    - name: Login to Docker Hub
      if: steps.registry.outputs.push == 'true'
      uses: docker/login-action@v3
      with:
        username: ${{ secrets.DOCKERHUB_USERNAME }}
        password: ${{ secrets.DOCKERHUB_TOKEN }}

    - name: Build image
      uses: docker/build-push-action@v6
      with:
        context: .
        file: ./Dockerfile
        platforms: linux/amd64,linux/arm64
        push: ${{ steps.registry.outputs.push == 'true' }}
        tags: ${{ steps.registry.outputs.push == 'true' && format('{0}:sha-{1}', steps.registry.outputs.image, github.sha) || '' }}
        cache-from: type=gha,scope=api-image
        cache-to: type=gha,mode=max,scope=api-image

要点：

  • PR 不推送：push=false 时 buildx 仍会完整执行构建并写入 GHA 缓存，能提前发现 Dockerfile 或 prune 问题。
  • main 才推送：同时要求 DOCKERHUB_USERNAME / DOCKERHUB_TOKEN 两个 secret 都存在；缺失时只打印 notice 并跳过推送，流水线不会变红。
  • 不新增 GitHub 权限：使用 Docker Hub 凭据而非 packages: write，工作流的 permissions: contents: read 保持不变。
  • 多架构：linux/amd64,linux/arm64；Dockerfile 的构建阶段固定在 $BUILDPLATFORM，只有运行阶段在 QEMU 下展开。
  • 缓存：type=gha 缓存只存 BuildKit 层，不包含镜像仓库凭据。

镜像内部结构、turbo prune / pnpm deploy / cache mount 的作用见 05 章。

多环境部署策略 ​

环境矩阵 ​

环境触发配置来源密钥来源入口
development本地 pnpm devshell 注入 process.env，模板见 apps/api/.env.example本地，不提交localhost:3000 / :5173
testpnpm test:unit / pnpm test:integration测试内显式构造对象，不读进程环境无随机端口，进程内
stagingpush main（Deploy 工作流）GitHub Environment staging + compose env_fileGitHub Environment secrets预发布域名
productionpush v* tag（Deploy 工作流）GitHub Environment production + 运行时注入Environment secrets + 审批人生产域名

配置优先级（从高到低）：

  1. 显式传给 loadConfig(env) 的对象（测试）；
  2. process.env：shell、docker run -e、CI env、compose environment；
  3. compose env_file（当前指向非敏感的 apps/api/.env.example）；
  4. schema 默认值（仅 development / test 允许省略 CORS_ORIGIN）。

应用不自动读取 .env：避免“本地能跑、容器里变量来源不明”的两套规则。.env.example 的职责是变量清单与本地 compose 的注入模板，真实密钥永远不进仓库、不进镜像。

部署工作流骨架 ​

.github/workflows/deploy.yml 只允许 main 分支与 v* tag 触发，PR 永不部署：

yaml
on:
  push:
    branches: [main]
    tags: ["v*"]

jobs:
  verify: # 始终执行：镜像构建 + 容器冒烟 + 负向配置校验
  preflight: # 读 vars.DEPLOY_ENABLED，决定是否继续

在没有云凭据时的行为：

  • verify 照常跑：docker build → 起容器 → 请求 /healthz、/readyz、/metrics → 再跑一次“故意缺变量”的负向校验；
  • 未设置仓库变量 DEPLOY_ENABLED=true 时，deploy-* 作业输出 notice 并跳过，流水线不会失败；
  • 镜像推送使用内置 secrets.GITHUB_TOKEN 到 GHCR，不需要额外云密钥；

回滚 runbook ​

原则：镜像回退优先。当前仓库没有数据库迁移，不要凭空引入迁移工具；配置类问题先回退环境变量，再回退镜像。

bash
# 1. 找到上一个可用镜像 tag（约定为 sha-<7 位 commit>）
docker images pnpm-monorepo-api

# 2. 只切换 tag，不重建业务配置
IMAGE_TAG=sha-<上一个> docker compose -f docker-compose.staging.yml up -d

# 3. 验证
curl -s localhost:3100/healthz    # version 应等于回退的 tag
curl -s localhost:3100/readyz     # status=ready，checks 全 ok
docker compose -f docker-compose.staging.yml logs api | tail -5

回滚检查单：

  • [ ] 旧镜像仍在本机/镜像仓库（保留最近 N 个 sha-* tag）
  • [ ] SERVICE_VERSION 与新 tag 一致，/healthz 能核对版本
  • [ ] /readyz 返回 200 后再恢复流量
  • [ ] 确认本次发布不包含不可逆的数据结构变更（当前没有）

结构化日志与 request id ​

  • 只使用 pino，输出单行 JSON；base 字段固定为 service / env / version，便于按环境过滤。
  • LOG_LEVEL 由环境配置控制；staging / production 默认 info，排查时可临时调到 debug。
  • x-request-id：合法（[A-Za-z0-9._-]{8,128}）则透传，否则生成 UUID，并写回响应头。访问日志携带同一个 id，客户端报错时可据此检索。
  • /healthz、/readyz、/health、/metrics 不写访问日志，避免探针按秒级频率刷屏。
  • 日志不记录请求体与请求头；pino redact 对 authorization、cookie、set-cookie、password、token、secret、apiKey 做兜底脱敏。

Web 侧的构建期环境变量 ​

apps/web 目前没有使用任何 VITE_*，所有配置都是构建期常量。需要明确的事实：

  • Vite 的 import.meta.env.VITE_* 在 vite build 时静态替换进产物，构建完成后无法通过环境变量修改；
  • 因此“一次构建、多环境部署”在默认模式下不成立，要么按环境分别构建，要么引入运行期配置；
  • 运行期注入的代价：需要在容器启动/页面加载时拉取 /config.json 或注入 window.__APP_CONFIG__，增加一次网络请求与失败兜底，且 CDN 缓存、CSP、首屏时序都要重新评估；
  • 本批不改变 web 的构建模式，保持多环境分别构建；等出现“同一镜像部署到多环境”的真实需求再评估运行期注入。

安全扫描 ​

yaml
# 教程依赖不会实时跟进补丁，audit 只作为可见信号，不阻断流水线。
# 阈值 high 与 registry 都写在根 package.json 的 audit 脚本里，CI 直接复用。
- name: Security audit
  continue-on-error: true
  run: pnpm run audit

- name: License check
  run: pnpm licenses list --json > licenses.json
bash
# 本地等价命令
pnpm run audit        # = pnpm audit --registry https://registry.npmjs.org --audit-level high

两个必须写对的地方：

  1. 用 pnpm run audit 而不是 pnpm audit。audit 同时是 pnpm 的内置命令，直接写 pnpm audit 会执行内置命令，不会命中仓库里的 audit 脚本，结果就是拿 .npmrc 的默认源去查审计接口。
  2. 脚本里显式指定 --registry https://registry.npmjs.org。仓库 .npmrc 用的是 registry.npmmirror.com，它没有实现 npm 的 audit 接口，直接跑会报 ERR_PNPM_AUDIT_ENDPOINT_NOT_EXISTS。审计只读取漏洞库、不改变安装源，所以不必为此改掉 .npmrc 的安装源。当前基线是 3 个漏洞：0 low / 3 moderate / 0 high / 0 critical。critical 已通过把 vitest 升到 4.x（不再依赖 tinypool 1.x）清除；high 的 zod 已通过迁移到 zod@4.6.7 消除；剩余 moderate 为 postcss-selector-parser、qs、minimatch，不触发 --audit-level high 门禁，详见仓库根目录 SECURITY.md 的「已知依赖风险」。

升级为阻断（去掉 continue-on-error）前需要评估：

  • 上游高危漏洞出现时，流水线会立即变红，但修复依赖往往需要先升级主版本，可能牵连框架与构建链；
  • 教程仓库的依赖以教学演示为主，不是生产服务；阻断会影响所有贡献者的 PR；
  • 若要阻断，建议只对 critical 级别阻断，high 保持可见，并配置 Dependabot 自动升级。

Action SHA 固定 ​

远程 action 一律写成 <owner>/<repo>@<40 位 commit SHA> # vN：

yaml
- uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262 # v4
  • tag 可以被上游作者强制移动，SHA 不行；行尾注释保留版本号，便于人工升级与 review。
  • annotated tag 陷阱：pnpm/action-setup、peaceiris/actions-gh-pages 的 v4 是 annotated tag，git/ref/tags/<tag> 返回的是 tag object 的 SHA，不能直接用于 uses。必须再取一次 git/tags/<sha>，用返回的 commit SHA（本仓库分别是 b906aff… 与 84c30a8…）。
  • 升级方式：Dependabot 会自动更新 SHA 与注释，无需手工盯版本。

Dependabot 分组策略 ​

pnpm workspace 只有一个根 lockfile，所以 npm 只扫描 /；按依赖类型分组，避免一次开出几十个 PR：

yaml
groups:
  production-dependencies:
    dependency-type: production
    update-types: [minor, patch]
  development-dependencies:
    dependency-type: development
    update-types: [minor, patch]
  • 同时最多 5 个 PR（open-pull-requests-limit），major 升级默认 ignore，交给人工。
  • github-actions 生态单独分组，保证 SHA 固定后仍能持续升级。

镜像扫描：Trivy（只报告） ​

deploy.yml 的 verify 作业在镜像构建、容器冒烟之后执行扫描：

yaml
- name: Scan image with Trivy (report only)
  continue-on-error: true # 只报告，不阻断部署
  run: |
    docker run --rm \
      -v /var/run/docker.sock:/var/run/docker.sock \
      -v "$PWD/artifacts/security:/out" \
      "${TRIVY_IMAGE}" image --scanners vuln --format json \
      --output /out/trivy-report.json "api-verify:${GITHUB_SHA}"
  • 扫描器镜像固定到「版本 + digest」（aquasec/trivy:0.58.2@sha256:…），防止上游 tag 被重写。
  • 下一步用 jq 统计 HIGH / CRITICAL 数量并写入 $GITHUB_STEP_SUMMARY；扫描失败只发 warning，不让 verify 变红。
  • 扫描的是本地构建出来的镜像，与后续推送到 GHCR 是同一份产物。

  • pnpm -r publish / changeset publish 会自动跳过 private: true 的包（apps/*）；三个库包 types / utils / ui 都会参与发布；
  • 真实发布开启 NPM_CONFIG_PROVENANCE=true，需要 workflow 的 id-token: write 权限。
yaml
# .github/workflows/release.yml（节选）
name: Release

on:
  push:
    branches: [main]
  workflow_dispatch:
    inputs:
      dry_run:
        type: boolean
        default: true
      tag:
        type: choice
        options: [latest, next, beta, canary]
        default: latest

permissions:
  contents: write # changeset publish 创建 git tag
  id-token: write # npm provenance

jobs:
  release:
    runs-on: ubuntu-latest
    env:
      DIST_TAG: ${{ inputs.tag || 'latest' }}
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: ./.github/actions/setup-pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Release check
        run: pnpm release:check

      - name: Decide publish mode
        id: mode
        env:
          NPM_TOKEN: ${{ secrets.NPM_TOKEN }}
          DRY_RUN: ${{ inputs.dry_run }}
        run: |
          if [ -z "$NPM_TOKEN" ]; then
            echo "publish=false" >> "$GITHUB_OUTPUT"
            echo "::notice title=skip publish::未配置 NPM_TOKEN，仅执行 dry-run 校验"
          elif [ "$DRY_RUN" = "true" ]; then
            echo "publish=false" >> "$GITHUB_OUTPUT"
            echo "::notice title=dry run::dry_run=true，跳过真实发布"
          else
            echo "publish=true" >> "$GITHUB_OUTPUT"
          fi

      - name: Publish dry-run
        if: steps.mode.outputs.publish != 'true'
        run: pnpm -r --filter "./packages/*" publish --dry-run --no-git-checks --tag "$DIST_TAG"

      - name: Configure npm auth
        if: steps.mode.outputs.publish == 'true'
        run: echo "//registry.npmjs.org/:_authToken=${NPM_TOKEN}" >> .npmrc
        env:
          NPM_TOKEN: ${{ secrets.NPM_TOKEN }}

      - name: Publish
        if: steps.mode.outputs.publish == 'true'
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
          NPM_CONFIG_REGISTRY: https://registry.npmjs.org/
          NPM_CONFIG_PROVENANCE: "true"
        run: pnpm release:tag "$DIST_TAG"

项目 .npmrc 默认指向 npmmirror，发布步骤通过 NPM_CONFIG_REGISTRY 覆盖为官方源；NPM_CONFIG_PROVENANCE 只在 CI 开启，本地发布不设置，避免缺少 OIDC 环境导致失败。
